# API + WebSocket server (ASGI)
daphne -b 0.0.0.0 -p ${PORT:-8000} boltbet_backend.asgi:application
# or, for local dev:
python manage.py runserver

# Background tasks. In deployment these run as in-process threads via the
# RUN_*_IN_PROCESS flags; each is also runnable standalone (e.g. if split out
# into its own service later).
python manage.py start_lightning_stream            # live ingest (SINGLETON)
python manage.py purge_strikes --hours 72          # raw-strike retention (scheduled)
python manage.py trim_country_strikes --keep 1000  # per-country trim (scheduled)
python manage.py resolve_strikes_bets --once       # settle pending Up/Down bets

# Migrations
python manage.py makemigrations lightning account
python manage.py migrate
